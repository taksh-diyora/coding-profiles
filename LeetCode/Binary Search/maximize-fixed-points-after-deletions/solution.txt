class Solution {
public:
    int maxFixedPoints(vector<int>& nums) {
        vector<pair<int, int>> prs;
        int n = nums.size();

        for(int i=0; i<n; i++){
            if(nums[i] <= i){
                prs.push_back({i - nums[i], nums[i]});
            }
        }

        sort(prs.begin(), prs.end());

        vector<int> lis;

        for(int i=0; i<prs.size(); i++){
            auto it = lower_bound(lis.begin(), lis.end(), prs[i].second);

            if(it == lis.end()){
                lis.push_back(prs[i].second);
            }else{
                *it = prs[i].second;
            }
        }

        return lis.size();
    }
};
