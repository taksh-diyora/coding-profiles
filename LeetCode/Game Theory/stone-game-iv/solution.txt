class Solution {
public:
    int dp[100010];
    bool solve(int n){
        if(n == 0) return 0;
        if(dp[n] != -1) return dp[n];

        bool ans = false;
        for(long long i=1; i*i <= n; i++){
            if(!solve(n-i*i)){
                ans = true;
                break;
            }
        }

        return dp[n] = ans;
    }
    bool winnerSquareGame(int n) {
        memset(dp, -1, sizeof(dp));
        return solve(n) == true;
    }
};
